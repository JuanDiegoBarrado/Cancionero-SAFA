\beginsong{Renacer en libertad}[]
    \beginverse
        \[Sol]No quiero una libertad que me hace libre frente a ti.
        \[Mim]No quiero quererte más por imponérmelo a mí mismo.
        \[Do]No quiero más regla que el amor.
        \[Re]No quiero más regla que tu voz.
    \endverse
    
    \beginverse
        \[Sol]Queremos estrenar vivir en la alegría,
        \[Mim]en comunidad, que nazca un nuevo día.
        \[Lam]En medio de problemas, luchas y acercamientos,
        \[Re]crecer hacia ti desde un nuevo pensamiento.
    \endverse

    \beginverse
        Para edificar la casa sobre roca, Fuera
        dudas, categorías, fuera las normas.
        El reino se construye confiando en mi voz,
        Sintiendo en la prueba "no temáis, soy yo".
    \endverse

    \beginverse
        Vive la libertad mirándote en los demás.
        Siente que creces al servir, que los demás
        renacen... Tú, con ellos, al fin.
    \endverse
\endsong
